# cryptography
# Used directly: Fernet encrypts the messages with the web app.
# CVE-2023-2650  - fixed in 41.0.0
# CVE-2023-4807  - fixed in 41.0.5
# CVE-2023-5678  - fixed in 41.0.7
# CVE-2023-49083 - fixed in 41.0.7
# CVE-2024-2511  - fixed in 43.0.1
# CVE-2024-4603  - fixed in 43.0.1
# CVE-2024-6119  - fixed in 43.0.1
# CVE-2024-12797 - fixed in 44.0.1
# CVE-2026-26007 - fixed in 46.0.5
# GHSA-537c-gmf6-5ccf - fixed in 48.0.1
# CVE-2026-69248 & CVE-2026-69249 - fixed in 49.0.0 - not upgraded, see below
# CVE-2026-69247 - fixed in 50.0.0 - not upgraded, see below
# Capped at 48.0.x: 49.0.0 dropped x86_64 macOS support (arm64 wheels only),
#   and the macOS bundles must include x86_64 (universal for Python 3.10+).
#   The CVEs fixed in 49.0.0 and 50.0.0 are not exploitable here: they only
#   affect cryptography's X.509 path validation (cryptography.x509.verification)
#   and PKCS#7 decryption, which nothing bundled uses. TLS goes through the
#   Python standard library ssl module.
cryptography~=48.0.1

#-------------------------------------------------------------------------------
# websockets
# Bundled as pure Python: install_source_only.sh removes the optional speedups
#   extension, and websockets falls back to its Python implementation.
# CVE-2018-1000518 - fixed in 5.0
# CVE-2021-33880 - fixed in 9.1
# One series per Python version: 15.x is the last supporting Python 3.9, 16.x
#   the last supporting Python 3.10 and 17.x requires Python 3.11+.
websockets>=15.0.1,<16 ; python_version <  "3.10"
websockets>=16.1.1,<17 ; python_version == "3.10"
websockets>=17.4,<18   ; python_version >= "3.11"
